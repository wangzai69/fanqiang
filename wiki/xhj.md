function toggleMenu() {
  const isOpen = nav.style.display === 'flex';

  nav.style.display = 'flex';
  overlay.style.display = 'block';
}

  nav.style.display = 'none';
  overlay.style.display = 'none';
}

// 点击遮罩层关闭菜单
overlay.addEventListener('click', closeMenu);

// 保持电脑端菜单始终显示

    nav.style.display = 'flex';

  } else {
    nav.style.display = 'none';
  }
}
